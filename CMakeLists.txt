cmake_minimum_required(VERSION 3.16)
project(gw-server VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../gw" CACHE PATH "GW source checkout")
set(GW_CORE_LIBRARY "${GW_ROOT}/libgw_server_core.a" CACHE FILEPATH "GW server-core static archive")
set(GW_NATIVE_LINK_FLAGS "" CACHE STRING "GW native link flags, supplied by make server")
set(OATPP_SOURCE_DIR "" CACHE PATH "Optional pinned oat++ source checkout")
if(OATPP_SOURCE_DIR)
  set(OATPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(OATPP_LINK_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  add_subdirectory("${OATPP_SOURCE_DIR}" oatpp EXCLUDE_FROM_ALL)
  set(OATPP_TARGET oatpp)
else()
  find_package(oatpp 1.3.0 CONFIG REQUIRED)
  set(OATPP_TARGET oatpp::oatpp)
endif()
if(NOT EXISTS "${GW_CORE_LIBRARY}" OR NOT GW_NATIVE_LINK_FLAGS)
  message(FATAL_ERROR "Build via make -C GW server GW_SERVER_DIR=${CMAKE_CURRENT_SOURCE_DIR}, or supply GW_CORE_LIBRARY and GW_NATIVE_LINK_FLAGS")
endif()
find_package(ZLIB REQUIRED)
add_executable(gw-server src/main.cpp)
set_target_properties(gw-server PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(gw-server PRIVATE "${GW_ROOT}/src" "${GW_ROOT}")
# Preserve paired Apple driver options when passing the GW link line through CMake.
string(REGEX REPLACE "-framework ([A-Za-z0-9_]+)" "-Wl,-framework,\\1" GW_NATIVE_LINK_FLAGS "${GW_NATIVE_LINK_FLAGS}")
string(REPLACE "-undefined dynamic_lookup" "" GW_NATIVE_LINK_FLAGS "${GW_NATIVE_LINK_FLAGS}")
separate_arguments(GW_LINK_FLAGS NATIVE_COMMAND "${GW_NATIVE_LINK_FLAGS}")
target_link_libraries(gw-server PRIVATE ${OATPP_TARGET} "${GW_CORE_LIBRARY}" ${GW_LINK_FLAGS} ZLIB::ZLIB)
if(MINGW)
  target_link_libraries(gw-server PRIVATE ws2_32)
endif()
install(TARGETS gw-server RUNTIME DESTINATION bin)
